import SwiftUI import Combine import UserNotifications // MARK: - Modelo struct Bill: Identifiable, Codable { var id = UUID() var name: String var dueDate: Date var isPaid: Bool = false // Quantos dias faltam (negativo = atrasada) var daysLeft: Int { let cal = Calendar.current let hoje = cal.startOfDay(for: Date()) let venc = cal.startOfDay(for: dueDate) return cal.dateComponents([.day], from: hoje, to: venc).day ?? 0 } var statusText: String { if isPaid { return "Paga" } let n = daysLeft if n < -1 { return "Venceu há \(-n) dias" } if n == -1 { return "Venceu há 1 dia" } if n == 0 { return "Vence hoje" } if n == 1 { return "Vence amanhã" } return "Vence em \(n) dias" } var statusColor: Color { if isPaid { return .green } if daysLeft < 0 { return .red } if daysLeft <= 3 { return .orange } return .secondary } } // MARK: - Dados, salvamento e notificações final class BillStore: ObservableObject { @Published var bills: [Bill] = [] { didSet { save() } } private let key = "contas-v1" init() { load() } // Não pagas primeiro, depois por data de vencimento var sorted: [Bill] { bills.sorted { a, b in if a.isPaid != b.isPaid { return !a.isPaid } return a.dueDate < b.dueDate } } // Aviso no topo da tela var banner: (text: String, color: Color)? { let open = bills.filter { !$0.isPaid } let late = open.filter { $0.daysLeft < 0 } if !late.isEmpty { let t = late.count == 1 ? "1 conta atrasada: \(late[0].name)" : "\(late.count) contas atrasadas" return (t, .red) } let soon = open.filter { $0.daysLeft <= 3 } if !soon.isEmpty { let t = soon.count == 1 ? "1 conta vence em breve: \(soon[0].name)" : "\(soon.count) contas vencem em breve" return (t, .orange) } return nil } func add(name: String, due: Date) { let bill = Bill(name: name, dueDate: due) bills.append(bill) schedule(bill) } func update(_ bill: Bill, name: String, due: Date) { guard let i = bills.firstIndex(where: { $0.id == bill.id }) else { return } bills[i].name = name bills[i].dueDate = due schedule(bills[i]) } func togglePaid(_ bill: Bill) { guard let i = bills.firstIndex(where: { $0.id == bill.id }) else { return } bills[i].isPaid.toggle() schedule(bills[i]) } func delete(_ bill: Bill) { cancel(bill) bills.removeAll { $0.id == bill.id } } // MARK: Salvar e carregar private func save() { if let data = try? JSONEncoder().encode(bills) { UserDefaults.standard.set(data, forKey: key) } } private func load() { guard let data = UserDefaults.standard.data(forKey: key), let saved = try? JSONDecoder().decode([Bill].self, from: data) else { return } bills = saved } // MARK: Notificações (às 9h, um dia antes e no dia do vencimento) private func ids(_ bill: Bill) -> [String] { ["\(bill.id.uuidString)-antes", "\(bill.id.uuidString)-dia"] } private func cancel(_ bill: Bill) { UNUserNotificationCenter.current().removePendingNotificationRequests(withIdentifiers: ids(bill)) } private func schedule(_ bill: Bill) { cancel(bill) if bill.isPaid { return } let cal = Calendar.current let dia = cal.startOfDay(for: bill.dueDate) let avisos: [(sufixo: String, deslocamento: Int, texto: String)] = [ ("antes", -1, "vence amanhã."), ("dia", 0, "vence hoje.") ] for aviso in avisos { guard let base = cal.date(byAdding: .day, value: aviso.deslocamento, to: dia), let quando = cal.date(bySettingHour: 9, minute: 0, second: 0, of: base), quando > Date() else { continue } let content = UNMutableNotificationContent() content.title = bill.name content.body = "A conta \(aviso.texto)" content.sound = .default let comps = cal.dateComponents([.year, .month, .day, .hour, .minute], from: quando) let trigger = UNCalendarNotificationTrigger(dateMatching: comps, repeats: false) let request = UNNotificationRequest( identifier: "\(bill.id.uuidString)-\(aviso.sufixo)", content: content, trigger: trigger ) UNUserNotificationCenter.current().add(request) } } } // MARK: - Tela principal enum FormMode: Identifiable { case new case edit(Bill) var id: String { switch self { case .new: return "new" case .edit(let b): return b.id.uuidString } } } struct BillsView: View { @StateObject private var store = BillStore() @State private var mode: FormMode? var body: some View { NavigationStack { List { if let banner = store.banner { Label(banner.text, systemImage: "bell.fill") .font(.subheadline.bold()) .foregroundStyle(banner.color) .listRowBackground(banner.color.opacity(0.12)) } if store.bills.isEmpty { Text("Nenhuma conta ainda. Toque em + para começar.") .foregroundStyle(.secondary) .frame(maxWidth: .infinity, alignment: .center) .padding(.vertical, 30) .listRowSeparator(.hidden) } ForEach(store.sorted) { bill in BillRow( bill: bill, onToggle: { store.togglePaid(bill) }, onEdit: { mode = .edit(bill) } ) .swipeActions(edge: .trailing, allowsFullSwipe: true) { Button(role: .destructive) { store.delete(bill) } label: { Label("Excluir", systemImage: "trash") } } } } .listStyle(.plain) .navigationTitle("Contas a vencer") .toolbar { ToolbarItem(placement: .primaryAction) { Button { mode = .new } label: { Label("Nova conta", systemImage: "plus") } } } .sheet(item: $mode) { m in switch m { case .new: BillForm(bill: nil) { nome, data in store.add(name: nome, due: data) } case .edit(let b): BillForm(bill: b) { nome, data in store.update(b, name: nome, due: data) } } } } } } // MARK: - Linha de cada conta struct BillRow: View { let bill: Bill let onToggle: () -> Void let onEdit: () -> Void var body: some View { HStack(spacing: 10) { VStack(alignment: .leading, spacing: 2) { Text(bill.name) .font(.headline) .strikethrough(bill.isPaid) .foregroundStyle(bill.isPaid ? Color.secondary : Color.primary) .lineLimit(1) HStack(spacing: 6) { Text(bill.statusText) .fontWeight(.bold) .foregroundStyle(bill.statusColor) Text(bill.dueDate.formatted(.dateTime.day().month(.abbreviated))) .foregroundStyle(.secondary) } .font(.footnote) .lineLimit(1) } Spacer(minLength: 4) Button(action: onEdit) { Image(systemName: "pencil") .foregroundStyle(.secondary) .frame(width: 32, height: 32) } .buttonStyle(.borderless) Button(action: onToggle) { Text(bill.isPaid ? "✓ Paga" : "Não paga") .font(.subheadline.bold()) .frame(minWidth: 70) .padding(.vertical, 7) .padding(.horizontal, 10) .foregroundStyle(bill.isPaid ? Color.white : Color.primary) .background(bill.isPaid ? Color.green : Color.clear, in: Capsule()) .overlay(Capsule().stroke(bill.isPaid ? Color.green : Color.primary, lineWidth: 2)) } .buttonStyle(.borderless) } .padding(.vertical, 2) } } // MARK: - Formulário (nova conta / editar conta) struct BillForm: View { @Environment(\.dismiss) private var dismiss @State private var name: String @State private var date: Date private let title: String private let onSave: (String, Date) -> Void init(bill: Bill?, onSave: @escaping (String, Date) -> Void) { _name = State(initialValue: bill?.name ?? "") _date = State(initialValue: bill?.dueDate ?? (Calendar.current.date(byAdding: .day, value: 3, to: Date()) ?? Date())) self.title = bill == nil ? "Nova conta" : "Editar conta" self.onSave = onSave } private var cleanName: String { name.trimmingCharacters(in: .whitespaces) } var body: some View { NavigationStack { Form { TextField("Nome da conta", text: $name) DatePicker("Vencimento", selection: $date, displayedComponents: .date) } .navigationTitle(title) #if os(iOS) .navigationBarTitleDisplayMode(.inline) #endif .toolbar { ToolbarItem(placement: .cancellationAction) { Button("Cancelar") { dismiss() } } ToolbarItem(placement: .confirmationAction) { Button("Salvar") { onSave(cleanName, date) dismiss() } .disabled(cleanName.isEmpty) } } } .presentationDetents([.medium]) } } // MARK: - Início do app @main struct ContasApp: App { init() { // Pede permissão para enviar avisos UNUserNotificationCenter.current().requestAuthorization(options: [.alert, .sound, .badge]) { _, _ in } } var body: some Scene { WindowGroup { BillsView() } } }
